## Riproduzione dello script di trascrizione effettivo

Salvare il blocco in `/tmp/lintra_task5_generate.py` ed eseguirlo dalla root
`lintra/`; riscrive soltanto le dieci tabelle. Richiede lo starter kit adiacente.

```python
import hashlib
import json
from pathlib import Path

kit = Path('../LINTRA_STARTER_KIT_FINAL_20261016')
pdf = 'SCHEMA_INPUTS/CAD_2024-06_SOURCE/2025-05-14_FLUSSO_ADI.pdf'
trace_keys = ('canonical_schema_version', 'schema_status', 'flow_id',
              'regulatory_version', 'trace_id', 'trace_name', 'record_length',
              'value_model', 'freeze_notes')
for flow, folder in [('sioc', 'SCHEMA_INPUTS/SIOC_2026-04-20_FROZEN'),
                     ('cad', 'CAD_REVISED_CANDIDATE')]:
    manifest = json.loads((kit / folder / 'manifest.json').read_text())
    outdir = Path('docs/normative') / flow
    outdir.mkdir(exist_ok=True)
    for entry in manifest['traces']:
        source = kit / folder / entry['file']
        trace = json.loads(source.read_text())
        files = [('received_trace', folder + '/' + entry['file']),
                 ('received_manifest', folder + '/manifest.json'),
                 ('received_flow', folder + '/flow_schema.json')]
        if flow == 'cad':
            files.append(('official_pdf_received', pdf))
        else:
            files.append(('received_freeze_report', folder + '/FREEZE_REPORT.md'))
        provenance = [{'role': role, 'kit_path': path,
                       'sha256': hashlib.sha256((kit / path).read_bytes()).hexdigest()}
                      for role, path in files]
        fields = [{key: field[key] for key in
                   ('id', 'name', 'source_id', 'physical', 'regulatory_metadata')
                   if key in field} for field in trace['fields']]
        table = {
            'document_status': 'DONE_UNVERIFIED',
            'purpose': 'T-P0-02 documentary transcription; human spot-check pending',
            'source_boundary': ('Received derived FROZEN JSON; official XLSX absent'
                                if flow == 'sioc' else
                                'Received RELEASE_CANDIDATE JSON; official PDF available'),
            'provenance': provenance,
            'source_manifest_metadata': manifest,
            'source_trace_metadata': {key: trace[key] for key in trace_keys},
            'fields': fields,
        }
        target = outdir / (entry['trace_id'] + '-fields.json')
        target.write_text(json.dumps(table, ensure_ascii=False, indent=2) + '\n')
        print(target, len(fields))
```

## Riproduzione della verifica effettiva

Salvare il blocco in `/tmp/lintra_task5_verify.py` ed eseguirlo dalla root
`lintra/`. Il controllo rilegge fonti e output, non legge né esegue export.
Il crop note usa offset testuali specifici del PDF ricevuto/estrazione corrente;
non è un parser PDF generale. Un diverso layout/tool può richiedere ispezione
prima di ripetere quel confronto; non cambiare i valori delle tabelle per farlo passare.

```python
import hashlib
import json
import re
import subprocess
from collections import Counter
from pathlib import Path

kit = Path('../LINTRA_STARTER_KIT_FINAL_20261016')
index = {item['path']: item for item in json.loads((kit / 'KIT_MANIFEST.json').read_text())['files']}
trace_keys = ('canonical_schema_version', 'schema_status', 'flow_id',
              'regulatory_version', 'trace_id', 'trace_name', 'record_length',
              'value_model', 'freeze_notes')
count = 0
checked_sources = set()
for flow, folder, total in [('sioc', 'SCHEMA_INPUTS/SIOC_2026-04-20_FROZEN', 135),
                            ('cad', 'CAD_REVISED_CANDIDATE', 83)]:
    manifest = json.loads((kit / folder / 'manifest.json').read_text())
    bundle = json.loads((kit / folder / 'flow_schema.json').read_text())
    assert {p.name for p in (Path('docs/normative') / flow).glob('*-fields.json')} == {
        e['trace_id'] + '-fields.json' for e in manifest['traces']}
    flow_count = 0
    types = Counter()
    for entry in manifest['traces']:
        trace = json.loads((kit / folder / entry['file']).read_text())
        table = json.loads((Path('docs/normative') / flow / (entry['trace_id'] + '-fields.json')).read_text())
        assert table['document_status'] == 'DONE_UNVERIFIED'
        assert table['source_manifest_metadata'] == manifest
        assert table['source_trace_metadata'] == {k: trace[k] for k in trace_keys}
        assert trace == next(t for t in bundle['traces'] if t['trace_id'] == entry['trace_id'])
        assert len(table['fields']) == len(trace['fields']) == entry['field_count']
        assert [f['id'] for f in table['fields']] == [f['id'] for f in trace['fields']]
        assert len({f['id'] for f in table['fields']}) == len(trace['fields'])
        assert all(set(f) <= {'id', 'name', 'source_id', 'physical', 'regulatory_metadata'}
                   for f in table['fields'])
        end = 0
        for row, field in zip(table['fields'], trace['fields']):
            expected_keys = {'id', 'name', 'physical', 'regulatory_metadata'}
            if 'source_id' in field:
                expected_keys.add('source_id')
            assert set(row) == expected_keys
            for key in expected_keys:
                assert row[key] == field[key], (flow, field['id'], key)
            assert {'source_id', 'source_obligation', 'source_note'} <= row['regulatory_metadata'].keys()
            assert isinstance(row['regulatory_metadata']['source_note'], str)
            physical = row['physical']
            assert type(physical['start']) is int and type(physical['length']) is int
            assert physical['start'] == end + 1 and physical['length'] > 0
            end = physical['start'] + physical['length'] - 1
            types[str(physical['source_type'])] += 1
        assert end == trace['record_length'] == entry['record_length']
        for source in table['provenance']:
            raw = (kit / source['kit_path']).read_bytes()
            expected = index[source['kit_path']]
            assert hashlib.sha256(raw).hexdigest() == source['sha256'] == expected['sha256']
            assert len(raw) == expected['bytes']
            checked_sources.add(source['kit_path'])
        count += len(trace['fields'])
        flow_count += len(trace['fields'])
        print(f'{flow}/{entry["trace_id"]}: {len(trace["fields"])} fields; range 1..{end}; gap=0 overlap=0; exact projection PASS; split=flow PASS')
    assert flow_count == total
    print(flow, 'total', flow_count, 'source_type values', dict(types))

pdf_path = kit / 'SCHEMA_INPUTS/CAD_2024-06_SOURCE/2025-05-14_FLUSSO_ADI.pdf'
extraction = subprocess.run(['pdftotext', '-layout', str(pdf_path), '-'],
                            check=True, capture_output=True, text=True).stdout
pages = extraction.split('\f')
assert len([page for page in pages if page.strip()]) == 7
row_pattern = re.compile(r'\b([1-5]\.\d+)\s+(?:X\s+)?(NO \(1\*\)|SI|NO|NBB|FAC)\s+.*?\s(\d+)\s+(\d+)\s+(AN|N|D|T)\s+(SI|NO|NBB|FAC)\b')
rows = {}
for page_number, page in enumerate(pages, 1):
    for line in page.splitlines():
        match = row_pattern.search(line)
        if match:
            fid, obligation, length, start, kind, nsis = match.groups()
            assert fid not in rows
            rows[fid] = (page_number, int(length), int(start), kind, obligation, nsis)
assert len(rows) == 83
# Note-column offsets for the five physical pages, zero-based text columns.
# This folds only comparison whitespace; the documentary JSON is never changed.
columns = [132, 163, 64, 120, 105]
note_pages = [re.sub(r'\s+', ' ', ' '.join(line[column:].strip() for line in page.splitlines())).strip()
              for column, page in zip(columns, pages)]
notes = 0
for path in sorted((kit / 'CAD_REVISED_CANDIDATE').glob('trace_*.json')):
    for field in json.loads(path.read_text())['fields']:
        physical, metadata = field['physical'], field['regulatory_metadata']
        assert rows[field['id']] == (metadata['source_page'], physical['length'],
               physical['start'], physical['source_type'], metadata['source_obligation'],
               metadata['source_nsis_obligation']), field['id']
        note = re.sub(r'\s+', ' ', metadata['source_note']).strip()
        assert note and note in note_pages[metadata['source_page'] - 1], field['id']
        notes += 1
assert count == 218 and notes == 83
print('CAD PDF: 7 pages; 83/83 physical rows + both obligations + source_page PASS')
print('CAD PDF: 83/83 full notes found after column crop and whitespace folding PASS')
print(f'TOTAL: 218 exact field projections; {len(checked_sources)} received source files match kit size/SHA; no patient/export reads')
```
